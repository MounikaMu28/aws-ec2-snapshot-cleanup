Pre-requisites:
1. AWS Account
2. IAM User with permissions to:
Create VPC
Create IAM roles
Create Lambda functions
Create EventBridge rules
3. Access to Ec2 Linux instance
4. AWS CLI installed
5. Terraform installed

Assumptions:
1. AWS Region: us-east-1
2. Lambda deletes only snapshots owned by the account
3. Snapshots older than 365 days are deleted
4. Lambda runs once per day
5. Lambda runs inside a private subnet
6. Internet access is available via AWS networking
7. Terraform is used for Infrastructure as Code

VPC & Subnet:
resource "aws_vpc" "main" {
  cidr_block = "10.0.0.0/16"
}

resource "aws_subnet" "private" {
  vpc_id     = aws_vpc.main.id
  cidr_block = "10.0.1.0/24"
}

Security Group:
resource "aws_security_group" "lambda_sg" {
  vpc_id = aws_vpc.main.id

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}


IAM role for Labmda:
resource "aws_iam_role" "lambda_role" {
  name = "lambda_snapshot_cleanup_role"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Effect = "Allow"
      Principal = {
        Service = "lambda.amazonaws.com"
      }
      Action = "sts:AssumeRole"
    }]
  })
}


IAM Policy:
resource "aws_iam_role_policy" "lambda_policy" {
  role = aws_iam_role.lambda_role.id

  policy = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Effect = "Allow"
      Action = [
        "ec2:DescribeSnapshots",
        "ec2:DeleteSnapshot",
        "logs:*"
      ]
      Resource = "*"
    }]
  })
}

Lambda Function:
resource "aws_lambda_function" "cleanup" {
  function_name = "delete-old-snapshots"
  role          = aws_iam_role.lambda_role.arn
  runtime       = "python3.10"
  handler       = "lambda_function.lambda_handler"
  filename      = "lambda.zip"

  vpc_config {
    subnet_ids         = [aws_subnet.private.id]
    security_group_ids = [aws_security_group.lambda_sg.id]
  }
}

EventBridge Rule (Daily Trigger):
resource "aws_cloudwatch_event_rule" "daily" {
  name                = "daily-snapshot-cleanup"
  schedule_expression = "rate(1 day)"
}

resource "aws_cloudwatch_event_target" "lambda_target" {
  rule = aws_cloudwatch_event_rule.daily.name
  arn  = aws_lambda_function.cleanup.arn
}

Step 4: Lambda Function Code
=================================================
-- lambda_function.py

import boto3
from datetime import datetime, timedelta, timezone

ec2 = boto3.client('ec2')

def lambda_handler(event, context):
    cutoff = datetime.now(timezone.utc) - timedelta(days=365)

    snapshots = ec2.describe_snapshots(OwnerIds=['self'])['Snapshots']

    for snap in snapshots:
        if snap['StartTime'] < cutoff:
            print(f"Deleting snapshot: {snap['SnapshotId']}")
            ec2.delete_snapshot(SnapshotId=snap['SnapshotId'])
